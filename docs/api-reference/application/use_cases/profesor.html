 59    @with_metrics("crear_profesor")
 60    def _execute_with_metrics(self, dto: CrearProfesorDTO) -> ProfesorDTO:
 61        """Ejecuta la lógica con métricas."""
 62        try:
 63            # 1. Validar que el nombre no esté duplicado
 64            profesor_existente = self.repository.find_by_nombre(dto.nombre_completo)
 65            if profesor_existente:
 66                raise ValidationError(
 67                    f"Ya existe un profesor con el nombre '{dto.nombre_completo}'"
 68                )
 69
 70            # 2. Crear Value Objects
 71            email = Email.from_optional(dto.email_corporativo) if dto.email_corporativo else None
 72            horas = HorasContrato(dto.horas_contrato)
 73            turno = Turno.from_string(dto.turno, dto.horas_manana, dto.horas_tarde)
 74
 75            # Zona preferida - crear desde ID si se proporciona
 76            zona_preferida = ZonaPreferida.sin_preferencia()
 77            if dto.zona_preferida_id:
 78                zona_preferida = ZonaPreferida.from_id(dto.zona_preferida_id)
 79
 80            # 3. Crear entidad de dominio
 81            entidad = ProfesorEntity(
 82                id=None,
 83                nombre_completo=dto.nombre_completo,
 84                email_corporativo=email,
 85                horas_contrato=horas,
 86                porcentaje_jornada=horas.porcentaje_jornada(),
 87                turno=turno,
 88                es_tutor=dto.tutor,  # DTO usa 'tutor', entidad usa 'es_tutor'
 89                fecha_inicio_guardias=dto.fecha_inicio_guardias,
 90                fecha_fin_guardias=dto.fecha_fin_guardias,
 91                zona_preferida=zona_preferida,
 92                dias_semana_permitidos=dto.dias_semana_permitidos,
 93                recreos_permitidos=dto.recreos_permitidos,
 94            )
 95
 96            # 7. Guardar en repositorio
 97            entidad_guardada = self.repository.save(entidad)
 98
 99            # 8. Invalidar cache de profesores
100            invalidate_profesores_cache()
101            logger.info(f"Cache invalidado tras crear: {entidad_guardada.nombre_completo}")
102
103            return self._entidad_to_dto(entidad_guardada)
104
105            logger.info(
106                "Profesor creado exitosamente",
107                profesor_id=entidad_guardada.id,
108                nombre=entidad_guardada.nombre_completo,
109            )
110
111            # 5. Convertir a DTO de salida
112            return self._entidad_to_dto(entidad_guardada)
113
114        except ValidationError:
115            self.session.rollback()
116            raise
117        except Exception as e:
118            self.session.rollback()
119            logger.error("Error al crear profesor", error=str(e))
120            raise ValidationError(f"Error al crear profesor: {e}") from e

133            tutor=entidad.es_tutor,  # Entidad usa 'es_tutor', DTO usa 'tutor'

 34        Inicializar el caso de uso.

 40
 41    @with_metrics("actualizar_profesor")
 42    def execute(self, profesor_id: int, data: ActualizarProfesorDTO) -> ProfesorDTO:
 43        """Execute wrapper with metrics applied via decorator below"""
 44        """
 45        Ejecutar la actualización de un profesor.
 46
 47        Args:
 48            profesor_id: ID del profesor a actualizar
 49            data: DTO con los datos a actualizar (campos opcionales)
 50
 51        Returns:
 52            ProfesorDTO con los datos actualizados del profesor
 53
 54        Raises:
 55            NotFoundError: Si no existe un profesor con ese ID
 56            BusinessLogicError: Si el nuevo nombre ya está en uso por otro profesor
 57        """
 58        # Buscar el profesor a actualizar
 59        profesor = self.session.query(Profesor).filter(Profesor.id == profesor_id).first()
 60
 61        if not profesor:
 62            raise NotFoundError(entity_type="Profesor", entity_id=profesor_id)
 63
 64        # Si se va a cambiar el nombre, verificar que no exista otro profesor con ese nombre
 65        if data.nombre_completo and data.nombre_completo != profesor.nombre_completo:
 66            try:
 67                validar_nombre_completo(data.nombre_completo)
 68            except ValueError as e:
 69                raise BusinessLogicError(str(e)) from e
 70
 71            profesor_existente = (
 72                self.session.query(Profesor)
 73                .filter(Profesor.nombre_completo == data.nombre_completo)
 74                .filter(Profesor.id != profesor_id)
 75                .first()
 76            )
 77
 78            if profesor_existente:
 79                raise BusinessLogicError(
 80                    f"Ya existe otro profesor con el nombre '{data.nombre_completo}'"
 81                )
 82
 83        # Validar otros campos si se proporcionan
 84        if data.horas_contrato is not None:
 85            try:
 86                validar_horas_contrato(data.horas_contrato)
 87            except ValueError as e:
 88                raise BusinessLogicError(str(e)) from e
 89
 90        if data.email_corporativo:
 91            try:
 92                validar_email(data.email_corporativo)
 93            except ValueError as e:
 94                raise BusinessLogicError(str(e)) from e
 95
 96        # Actualizar campos si se proporcionan
 97        if data.nombre_completo is not None:
 98            profesor.nombre_completo = data.nombre_completo
 99
100        if data.email_corporativo is not None:
101            profesor.email_corporativo = data.email_corporativo
102
103        if data.horas_contrato is not None:
104            profesor.horas_contrato = data.horas_contrato
105            # Recalcular porcentaje de jornada
106            profesor.porcentaje_jornada = data.horas_contrato / 25.0
107
108        if data.turno is not None:
109            profesor.turno = data.turno
110
111        if data.horas_manana is not None:
112            profesor.horas_manana = data.horas_manana
113
114        if data.horas_tarde is not None:
115            profesor.horas_tarde = data.horas_tarde
116
117        if data.tutor is not None:
118            profesor.tutor = data.tutor
119
120        if data.fecha_inicio_guardias is not None:
121            profesor.fecha_inicio_guardias = data.fecha_inicio_guardias
122
123        if data.fecha_fin_guardias is not None:
124            profesor.fecha_fin_guardias = data.fecha_fin_guardias
125
126        if data.dias_semana_permitidos is not None:
127            profesor.dias_semana_permitidos = json.dumps(data.dias_semana_permitidos)
128
129        # Actualizar recreos_permitidos incluso si es un diccionario vacío
130        # Esto permite limpiar las restricciones
131        if data.recreos_permitidos is not None:
132            if isinstance(data.recreos_permitidos, dict):
133                # Guardar diccionario vacío como string vacío
134                if not data.recreos_permitidos:
135                    profesor.recreos_permitidos = ""
136                else:
137                    profesor.recreos_permitidos = json.dumps(data.recreos_permitidos)
138            else:
139                profesor.recreos_permitidos = json.dumps(data.recreos_permitidos)
140
141        try:
142            self.session.commit()
143            self.session.refresh(profesor)
144
145            # Invalidar cache de profesores
146            invalidate_profesores_cache()
147            logger.info(
148                f"Profesor actualizado y cache invalidado: "
149                f"{profesor.nombre_completo} (ID: {profesor.id})"
150            )
151
152            return self._convertir_a_dto(profesor)
153
154        except Exception as e:
155            self.session.rollback()
156            raise BusinessLogicError(f"Error al actualizar el profesor: {str(e)}") from e
