12class ListarPerfilesUseCase:
13    """Lista todos los perfiles de usuario del sistema."""
14
15    def __init__(self, user_auth: UserAuth):
16        self.user_auth = user_auth
17
18    def execute(self) -> List[PerfilDTO]:
19        """
20        Obtiene la lista completa de perfiles.
21
22        Returns:
23            Lista de PerfilDTO con todos los perfiles del sistema
24        """
25        perfiles = []
26        current_user = get_current_user_id()
27
28        for username, data in self.user_auth.users.items():
29            # Verificar si tiene BD
30            tiene_bd = user_has_database(username)
31
32            # Verificar si tiene logo
33            logo_path = Path("imagenes") / f"{username}.png"
34            tiene_logo = logo_path.exists()
35
36            # Es el usuario actual
37            es_actual = username == current_user
38
39            perfil = PerfilDTO(
40                username=username,
41                email=data.get("email", ""),
42                tiene_bd=tiene_bd,
43                tiene_logo=tiene_logo,
44                es_actual=es_actual,
45                ruta_logo=str(logo_path) if tiene_logo else None,
46            )
47            perfiles.append(perfil)
48
49        # Ordenar: usuario actual primero, luego alfabético
50        perfiles.sort(key=lambda p: (not p.es_actual, p.username.lower()))
51
52        return perfiles

12    """Crea un nuevo perfil de usuario con su base de datos."""

14    def __init__(self, user_auth: UserAuth):

16
17    def execute(self, dto: CrearPerfilDTO) -> PerfilDTO:
18        """
19        Crea un nuevo perfil.
20
21        Args:
22            dto: DTO con los datos del nuevo perfil
23
24        Returns:
25            PerfilDTO del perfil creado
26
27        Raises:
28            ValidationError: Si los datos son inválidos o el usuario ya existe
29        """
30        # Validaciones
31        if not dto.username or not dto.username.strip():
32            raise ValidationError("El nombre de usuario es obligatorio")
33
34        if not dto.email or not dto.email.strip():
35            raise ValidationError("El email es obligatorio")
36
37        if "@" not in dto.email or "." not in dto.email:
38            raise ValidationError("El email no es válido")
39
40        if not dto.password or len(dto.password) < 4:
41            raise ValidationError("La contraseña debe tener al menos 4 caracteres")
42
43        username = dto.username.strip()
44        email = dto.email.strip()
45
46        # Verificar que no exista
47        if username in self.user_auth.users:
48            raise ValidationError(f"Ya existe un usuario con el nombre '{username}'")
49
50        # Crear usuario en UserAuth con register_user
51        if not self.user_auth.register_user(username, dto.password, email):
52            raise ValidationError("No se pudo crear el usuario")
53
54        # Crear base de datos
55        if not create_user_database(username):
56            # Revertir creación de usuario con unregister_user
57            self.user_auth.unregister_user(username)
58            raise ValidationError("No se pudo crear la base de datos del usuario")
59
60        # Retornar DTO del perfil creado
61        current_user = get_current_user_id()
62        return PerfilDTO(
63            username=username,
64            email=email,
65            tiene_bd=True,
66            tiene_logo=False,
67            es_actual=username == current_user,
68            ruta_logo=None,
69        )

class EliminarPerfilUseCase: 
11class EliminarPerfilUseCase:
12    """Elimina un perfil de usuario y todos sus datos asociados."""
13
14    def __init__(self, user_auth: UserAuth):
15        self.user_auth = user_auth
16
17    def execute(self, username: str) -> bool:
18        """
19        Elimina un perfil y sus datos.
20
21        Args:
22            username: Nombre del usuario a eliminar
23
24        Returns:
25            True si se eliminó correctamente
26
27        Raises:
28            NotFoundError: Si el perfil no existe
29            ValidationError: Si se intenta eliminar el perfil actual
30        """
31        # Validar que no sea el usuario actual
32        current_user = get_current_user_id()
33        if username == current_user:
34            raise ValidationError("No puedes eliminar tu propio perfil mientras estás logueado")
35
36        # Verificar que existe
37        if username not in self.user_auth.users:
38            raise NotFoundError(f"No existe el usuario '{username}'")
39
40        # Eliminar base de datos
41        delete_user_database(username)
42
43        # Eliminar logo si existe
44        logo_path = Path("imagenes") / f"{username}.png"
45        if logo_path.exists():
46            logo_path.unlink()
47
48        # Eliminar del sistema de autenticación usando unregister_user
49        self.user_auth.unregister_user(username)
50
51        return True

class CambiarPasswordUseCase: 
12class CambiarPasswordUseCase:
13    """Cambia la contraseña del usuario actual."""
14
15    def __init__(self, user_auth: UserAuth):
16        self.user_auth = user_auth
17
18    def execute(self, dto: CambiarPasswordDTO) -> bool:
19        """
20        Cambia la contraseña del usuario.
21
22        Args:
23            dto: DTO con las contraseñas
24
25        Returns:
26            True si se cambió correctamente
27
28        Raises:
29            ValidationError: Si las validaciones fallan
30        """
31        # Solo se puede cambiar la contraseña del usuario actual
32        current_user = get_current_user_id()
33        if dto.username != current_user:
34            raise ValidationError("Solo puedes cambiar tu propia contraseña")
35
36        # Validar que las contraseñas nuevas coinciden
37        if dto.password_nueva != dto.password_confirmacion:
38            raise ValidationError("Las contraseñas nuevas no coinciden")
39
40        # Validar longitud mínima
41        if len(dto.password_nueva) < 4:
42            raise ValidationError("La contraseña debe tener al menos 4 caracteres")
43
44        # Verificar contraseña actual usando authenticate
45        if not self.user_auth.authenticate(dto.username, dto.password_actual):
46            raise ValidationError("La contraseña actual es incorrecta")
47
48        # Cambiar contraseña
49        if dto.username in self.user_auth.users:
50            # Hashear nueva contraseña (igual que en UserAuth)
51            password_hash = hashlib.sha256(dto.password_nueva.encode()).hexdigest()
52            self.user_auth.users[dto.username]["password_hash"] = password_hash
53            self.user_auth._save_users()
54            return True
55
56        raise ValidationError("Error al cambiar la contraseña")

class ActualizarLogoUseCase: 
11class ActualizarLogoUseCase:
12    """Actualiza el logo corporativo de un perfil."""
13
14    def __init__(self, user_auth: UserAuth):
15        self.user_auth = user_auth
16
17    def execute(self, username: str, ruta_imagen: str) -> str:
18        """
19        Actualiza el logo de un perfil.
20
21        Args:
22            username: Nombre del usuario
23            ruta_imagen: Ruta al archivo de imagen
24
25        Returns:
26            Ruta donde se guardó el logo
27
28        Raises:
29            NotFoundError: Si el perfil no existe
30            ValidationError: Si la imagen no es válida
31        """
32        # Verificar que existe el perfil
33        if username not in self.user_auth.users:
34            raise NotFoundError(f"No existe el usuario '{username}'")
35
36        # Verificar que el archivo existe
37        archivo_origen = Path(ruta_imagen)
38        if not archivo_origen.exists():
39            raise ValidationError("El archivo de imagen no existe")
40
41        # Verificar extensión válida
42        extensiones_validas = {".png", ".jpg", ".jpeg", ".bmp"}
43        if archivo_origen.suffix.lower() not in extensiones_validas:
44            raise ValidationError(
45                f"Formato no válido. Use: {', '.join(extensiones_validas)}"
46            )
47
48        # Crear carpeta si no existe
49        carpeta_imagenes = Path("imagenes")
50        carpeta_imagenes.mkdir(exist_ok=True)
51
52        # Copiar imagen con nombre del usuario (siempre .png)
53        destino = carpeta_imagenes / f"{username}.png"
54
55        try:
56            shutil.copy(archivo_origen, destino)
57            return str(destino)
58        except Exception as e:
59            raise ValidationError(f"Error al copiar la imagen: {str(e)}")
