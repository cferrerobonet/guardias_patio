162    def puede_asignar_guardia(
163        self,
164        fecha: date,
165        turno_recreo: str,
166        numero_recreo: int,
167        zona_id: Optional[int] = None
168    ) -> tuple[bool, Optional[str]]:
169        """
170        Verifica de forma completa si se puede asignar una guardia.
171
172        Args:
173            fecha: Fecha de la guardia
174            turno_recreo: Turno del recreo
175            numero_recreo: Número del recreo
176            zona_id: ID de la zona (opcional)
177
178        Returns:
179            Tupla (puede_asignar, razon_si_no_puede)
180
181        Examples:
182            >>> profesor = ProfesorEntity(...)
183            >>> puede, razon = profesor.puede_asignar_guardia(date.today(), "mañana", 1)
184            >>> if puede:
185            ...     profesor.asignar_guardia()
186        """
187        # Verificar fecha
188        try:
189            if not self.puede_hacer_guardia_en_fecha(fecha):
190                return False, f"Fecha no disponible: {fecha}"
191        except ProfesorAusenteError as e:
192            return False, str(e)
193
194        # Verificar turno
195        if not self.puede_hacer_guardia_en_turno(turno_recreo):
196            return False, f"No trabaja en turno {turno_recreo}"
197
198        # Verificar recreo
199        if not self.puede_hacer_guardia_en_recreo(numero_recreo):
200            return False, f"No puede hacer guardia en recreo {numero_recreo}"
201
202        # Verificar máximo de guardias por día
203        if self.guardias_asignadas_dia >= settings.max_guardias_por_profesor_dia:
204            return False, (
205                f"Ya tiene {self.guardias_asignadas_dia} guardias hoy "
206                f"(máx: {settings.max_guardias_por_profesor_dia})"
207            )
208
209        # Verificar zona preferida (si tiene)
210        if self.zona_preferida.tiene_preferencia and zona_id:
211            if not self.zona_preferida.coincide_con(zona_id):
212                # Permitir pero con advertencia
213                return True, f"Zona {zona_id} no es su preferida ({self.zona_preferida.zona_id})"
214
215        return True, None
