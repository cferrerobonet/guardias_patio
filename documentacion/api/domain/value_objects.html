 49        >>> turno = Turno(TurnoEnum.MANANA)
 50        >>> print(turno.value)
 51        mañana

 55        >>> turno_mixto.es_completo
 56        True

 58
 59    value: TurnoEnum
 60    horas_manana: Optional[float] = None

 63    # Turnos válidos desde configuración
 64    TURNOS_VALIDOS: ClassVar[list[str]] = settings.turnos_validos
 65
 66    def __post_init__(self) -> None:
 67        """Valida el turno después de la inicialización."""
 68        # Validar que el turno esté en los válidos
 69        if self.value.value not in self.TURNOS_VALIDOS:
 70            raise ValidationError(
 71                message=(
 72                    f"Turno '{self.value.value}' no está en turnos válidos: "
 73                    f"{self.TURNOS_VALIDOS}"
 74                )
 75            )
 76
 77        # Si es turno mixto, validar que tenga horas asignadas
 78        if self.value == TurnoEnum.MIXTO:
 79            if self.horas_manana is None and self.horas_tarde is None:
 80                raise ValidationError(
 81                    message="Turno mixto requiere especificar horas_manana y/o horas_tarde"
 82                )
 83
 84            # Validar que las horas sean positivas
 85            if self.horas_manana is not None and self.horas_manana < 0:
 86                raise ValidationError(
 87                    message=f"horas_manana no puede ser negativa: {self.horas_manana}"
 88                )
 89
 90            if self.horas_tarde is not None and self.horas_tarde < 0:
 91                raise ValidationError(
 92                    message=f"horas_tarde no puede ser negativa: {self.horas_tarde}"
 93                )
 94
 95        # Si no es turno mixto, no debería tener horas específicas
 96        elif self.horas_manana is not None or self.horas_tarde is not None:
 97            raise ValidationError(
 98                message=(
 99                    f"Turno '{self.value.value}' no debe tener horas_manana/horas_tarde "
100                    "especificadas"
101                )
102            )
